{# Pagination compacte et scalable : seules quelques pages autour de la page courante sont affichées, avec des ellipses vers la première et la dernière. À importer dans un template : {% import "_pagination.html" as pg %}. #} {% macro pagination(page, pages, endpoint, link_args={}) %} {% if pages > 1 %} {% set start = [1, page - 2]|max %} {% set end = [pages, page + 2]|min %}
    • {% if start > 1 %}
  • 1
    • {% if start > 2 %}
  • …
    • {% endif %} {% endif %} {% for p in range(start, end + 1) %}
  • {{ p }}
    • {% endfor %} {% if end < pages %} {% if end < pages - 1 %}
  • …
    • {% endif %}
  • {{ pages }}
    • {% endif %}
{% endif %} {% endmacro %}
